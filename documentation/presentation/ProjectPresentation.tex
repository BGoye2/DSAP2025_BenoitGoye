    \begin{frame}{Statistical Validation Framework}
        \framesubtitle{Data \& Methodology}

        \begin{block}{\textbf{1.} Bootstrap Confidence Intervals}
            Train each model \textbf{100 times} on bootstrap samples. Compute 95\% CI. Parallel processing (\textbf{6$\times$} speedup).
        \end{block}

        \begin{block}{\textbf{2.} Permutation Importance Tests}
            Randomly permute features (\textbf{50} permutations). Measure drop in $R^2$. One-sample t-test ($\alpha = 0.05$).
        \end{block}

        \begin{block}{\textbf{3.} Cross-Model Consistency}
            Spearman rank correlations between models. High $\rho > 0.7$ $\rightarrow$ robust rankings.
        \end{block}

        \vspace{0.2cm}
        \begin{alertblock}{\textbf{Goal}}
            Ensure feature importance rankings are \textbf{statistically robust} and \textbf{consistent}
        \end{alertblock}
    \end{frame}
